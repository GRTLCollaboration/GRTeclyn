amr.check_int = -1
# Number of level-0 time steps between plot files; -1 disables output
amr.plot_int = -1

#################################################

bh1.mass = 0.5

#################################################
# Grid parameters

amr.n_cell = 64 64 64
geometry.prob_extent = 8.0 8.0 8.0

# We only need the first timestep's data, so never regrid
amr.max_level = 1
amr.regrid_int = 0

#################################################
# Boundary condition parameters

# Periodic directions: 0 = false, 1 = true
geometry.is_periodic = 0 0 0
# Options are currently FIRST_ORDER_EXTRAPOLATION_BC, SOMMERFELD_BC,
# or REFLECTIVE_BC
boundary.hi_condition = "SOMMERFELD_BC" "SOMMERFELD_BC" "SOMMERFELD_BC"

#################################################
# Evolution parameters

evolution.max_steps = 1

#################################################
# AH finder pseudo-time integrator (AMReX TimeIntegrator)
# type: ForwardEuler or RungeKutta
integration.type = RungeKutta
# Butcher tableau for RungeKutta (2=Trapezoid, 3=SSPRK3, 4=RK4); ignored for
# ForwardEuler.
integration.rk.type = 4

#################################################
# AH finder parameters

test.num_particles = 64
particle_interpolator.verbosity = false

# The surface radius h is evolved as a damped wave,
#   h_dot = v - eta * h,  v_dot = -c^2 * Theta,
# until the inf-norm of the expansion Theta falls below the tolerance.

# Damping coefficient
ah_finder.eta = 3.0
# Wave speed
ah_finder.c = 1.0
# Convergence threshold on the inf-norm of Theta
ah_finder.tolerance = 1e-4
# Target growth factor of the adaptive pseudo-timestep: dt is scaled each step
# by r * theta_old / theta_new
ah_finder.r = 1.15
