#################################################
# AH finder parameters

test.num_particles = 2048
particle_interpolator.verbosity = false

# The surface radius h is evolved by implicit pseudo-transient continuation of
#   h_dot = -Theta,  i.e. (J + I/dt) delta_h = -Theta each step,
# until the inf-norm of the expansion Theta falls below the tolerance (or
# max_iter iterations have been taken, or dt has sat at its minimum for
# max_stall_iter consecutive iterations).

# Convergence threshold on the inf-norm of Theta
ah_finder.tolerance = 1e-4
# Target growth factor of the adaptive pseudo-timestep: dt is scaled each step
# by r * theta_old / theta_new
ah_finder.r = 1.15
# Pseudo-timestep for the first step; adapted thereafter
ah_finder.initial_dt = 1e-2
# Maximum number of pseudo-time iterations before giving up
ah_finder.max_iter = 1000
# Maximum number of consecutive iterations at the minimum pseudo-timestep
# before the solve is treated as stalled
ah_finder.max_stall_iter = 50
