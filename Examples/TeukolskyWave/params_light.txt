#################################################
# Teukolsky Wave parameters

# Initial Data is provided via the Eppley packet construction.
# Implemented (parity, magnetic) pairs: (even, 0), (even, 2), (odd, 2)
teukolsky_wave.parity = even
teukolsky_wave.magnetic = 0
# Control parameters for the seed function
# The code has been designed for the scenario
# regularize_r << sigma << radial_offset.
# The amplitude should be sufficiently small, so that we remain close to the
# linear regime for the low-resolution example.
teukolsky_wave.sigma = 1.5
teukolsky_wave.amplitude = 0.01
teukolsky_wave.radial_offset = 5.
# A small parameter that regularizes the initial data close to the origin.
teukolsky_wave.regularize_r = 0.1
