# Wormhole Initial Data parameters

# How to switch between the two supported initial-data setups
# ----------------------------------------------------------
# This example supports TWO metric realisations, controlled by `wormhole_metric_type`.
#
# (A) Single-throat Morris–Thorne / Ellis–Bronnikov (recommended for x/y/z symmetry)
#     wormhole_metric_type = 1
#     - Uses ONLY `wormhole_centerA` as the throat location (centerB is ignored).
#     - Typical symmetry setup (octant):
#         center      = 0.0 0.0 0.0
#         lo_boundary = 2 2 2
#     - Typical extraction:
#         extraction_center = 0.0 0.0 0.0
#
# (B) Two-mouth isotropic superposition (binary-like)
#     wormhole_metric_type = 0
#     - Uses BOTH `wormhole_centerA` and `wormhole_centerB`.
#     - Typical symmetry setup (reflect only on y=0, z=0 planes):
#         center      = L_full/2  0.0 0.0
#         lo_boundary = 1 2 2
#     - Typical extraction:
#         extraction_center = center (same as grid center)
#     - You must also set:
#         wormhole_throat_radius_A / wormhole_throat_radius_B
#         wormhole_centerA / wormhole_centerB
#     (For this mode, keep x as a full axis—no symmetry at x=0—so both mouths exist.)
#
wormhole_metric_type = 1
# Initial lapse at t=0:
# - 0: alpha = 1
# - 1: alpha = sqrt(chi) (pre-collapsed)
# - 2: alpha = 1 - 3*log(chi) (N=2, 1+log-inspired using gamma=chi^{-3})
wormhole_initial_lapse_type = 0

# Kick model:
#   K(r,theta) = [A0 + A2 * (3 cos^2(theta) - 1)] * exp(-r^2 / sigma^2)
# where:
# - A0 is the monopole / coherent compression term
# - A2 is the quadrupole term used to break spherical symmetry
# - the legacy parameter `wormhole_k_amplitude` is now just a backward-compatible
#   alias for the quadrupole amplitude
#
# Recommended usage for collapse studies:
# - choose A0 to drive the global pinch-off
# - keep |A2| << |A0| so the quadrupole perturbs the collapse without dominating it
# - keep width = 1.0 for now and scan amplitudes gradually
#was 0.5 initally
wormhole_throat_radius = 0.5 
wormhole_centerA = 0.0 0.0 0.0
# Kick coefficients in:
#   K(r,theta) = [A0 + A2 * (3 cos^2(theta) - 1)] * exp(-r^2 / sigma^2)
# where:
# - A0 = wormhole_k_monopole_amplitude
#   coherent spherically symmetric compression / expansion term
# - A2 = wormhole_k_quadrupole_amplitude
#   symmetry-breaking Y20 term used to generate non-spherical dynamics / Psi4
#
# Collapse attempt (supercritical kick, stabilized numerics):
# - A0 drives the main collapse dynamics; A2 just breaks symmetry for Psi4
# - in this code, positive K is the branch that drives the 1+log lapse downward
# - use a narrower, stronger monopole to recover the collapsing branch
